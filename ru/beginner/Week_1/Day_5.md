```python
from art import tprint
import sys

def get_style():
    """Получает выбор стиля от пользователя"""
    
    print("Выберите стиль ASCII-арта:")
    print("1. block (блочный)")
    print("2. standard (стандартный)")
    print("3. slant (наклонный)")
    print("4. banner (баннер)")
    print()
    
    choice = input("Введите номер стиля (1-4): ").strip()

    }

def save_to_file(ascii_text):
    """Сохраняет ASCII-арт в файл"""
    
    try:
        with open("result.txt", "w", encoding="utf-8") as f:
            f.write(ascii_text)
        return True
    except Exception as e:
        print(f"❌ Ошибка при сохранении: {e}")
        return False

def main():
    """Главная функция программы"""
    
    print("=" * 40)
    print("       ASCII Арт Генератор")
    print("=" * 40)
    print()
    
    # Запрашиваем имя

        name = input("Введите ваше имя: ").strip()
        
        if name:
            break
        print("❌ Имя не может быть пустым! Попробуйте снова.")
    
    print()
    
    # Получаем стиль
    style = get_style()
    
    print()
    print(f"Создаём '{name}' в стиле '{style}'...")
    print()
    
    # Генерируем ASCII-арт (в строку)
    from art import art
    ascii_result = art(name, font=style)
    print(ascii_result)

    print()

    if save_choice in ["да", "yes", "y", "д"]:
        if save_to_file(ascii_result):
            print("✅ Сохранено в файл: result.txt")
    else:
        print("Ок, не сохраняем.")
    
    print()
    print("Спасибо за использование! 👋")

if __name__ == "__main__":
    main()
```

## Часть 4: Альтернатива без библиотек

Если не хочешь устанавливать библиотеку `art`, делегируй агенту версию без зависимостей:

### Задача:

```text
Напиши версию без внешних библиотек.
Создай несколько предустановленных шаблонов букв вручную.
Пользователь может ввести только заглавные латинские буквы.
```

### Пример кода:

```python
# Простые ASCII шаблоны для букв
ASCII_LETTERS = {
    "A": [
        "  AAA  ",
        " A   A ",
        "AAAAAAA",
        "A     A",
        "A     A"
    ],
    "B": [
        "BBBB  ",
        "B   B ",
        "BBBB  ",
        "B   B ",
        "BBBB  "
    ],
    "C": [
        " CCCC ",
        "C     ",
        "C     ",
        "C     ",
        " CCCC "
    ],
    # ... остальные буквы
    " ": [
        "   ",
        "   ",
        "   ",
        "   ",
        "   "
    ]
}

def generate_ascii(text):
    """Генерирует ASCII-арт из текста"""
    
    text = text.upper()
    lines = [""] * 5  # 5 строк высота букв
    
    for char in text:
        if char in ASCII_LETTERS:
            char_lines = ASCII_LETTERS[char]
            for i, line in enumerate(char_lines):
                lines[i] += line + " "
    
    return "\n".join(lines)

def main():
    print("=== ASCII Арт Генератор (без библиотек) ===")
    print()
    
    name = input("Введите имя (только латинские буквы): ").strip()
    
    if not name:
        print("Имя не может быть пустым!")
        return
    
    print()
    print(generate_ascii(name))

if __name__ == "__main__":
    main()
```
